  26def _first_or_none(res) -> dict | None:
  27    """PostgREST returns 406 when .maybe_single() gets 0 rows (single-object Accept). Use limit(1) + this."""
  28    if res is None:
  29        return None
  30    data = getattr(res, "data", None)
  31    if data is None:
  32        return None
  33    if isinstance(data, list):
  34        return data[0] if data else None
  35    if isinstance(data, dict):
  36        return data
  37    return None

 158def delete_project(project_id: str):
 159    """Delete a project and its related data (tasks, sessions, commands, logs)."""
 160    sb = get_sb()
 161    # Delete in dependency order.
 162    # Terminal logs → commands → sessions
 163    sessions_res = sb.table("terminal_sessions").select("id").eq("project_id", project_id).execute()
 164    session_ids = [s["id"] for s in (sessions_res.data or [])]
 165    # Conversation turns + state (must precede terminal_commands due to FK on command_id)
 166    try:
 167        sb.table("conversation_turns").delete().eq("project_id", project_id).execute()
 168    except Exception:
 169        pass
 170    try:
 171        sb.table("conversation_state").delete().eq("project_id", project_id).execute()
 172    except Exception:
 173        pass
 174    if session_ids:
 175        cmds_res = sb.table("terminal_commands").select("id").in_("session_id", session_ids).execute()
 176        cmd_ids = [c["id"] for c in (cmds_res.data or [])]
 177        if cmd_ids:
 178            # Also clean up any conversation_turns referencing these commands
 179            try:
 180                sb.table("conversation_turns").delete().in_("command_id", cmd_ids).execute()
 181            except Exception:
 182                pass
 183            sb.table("terminal_logs").delete().in_("command_id", cmd_ids).execute()
 184            sb.table("terminal_commands").delete().in_("id", cmd_ids).execute()
 185        sb.table("terminal_sessions").delete().in_("id", session_ids).execute()
 186    # Tasks + agent executions
 187    tasks_res = sb.table("tasks").select("id").eq("project_id", project_id).execute()
 188    task_ids = [t["id"] for t in (tasks_res.data or [])]
 189    if task_ids:
 190        sb.table("agent_executions").delete().in_("task_id", task_ids).execute()
 191    sb.table("tasks").delete().eq("project_id", project_id).execute()
 192    # Instances (local agent registrations)
 193    sb.table("instances").delete().eq("project_id", project_id).execute()
 194    # Device project links
 195    sb.table("device_project_links").delete().eq("project_id", project_id).execute()
 196    # Cursor context
 197    sb.table("cursor_context_snapshots").delete().eq("project_id", project_id).execute()
 198    # Project itself
 199    sb.table("projects").delete().eq("id", project_id).execute()
 200    logger.debug("delete_project project_id=%s", project_id)

 222def upsert_project_by_name(*, user_id: str, name: str, file_path: str | None = None) -> dict:
 223    """Return an existing project by name or create it; update file_path if provided."""
 224    existing = get_project_by_name(user_id, name)
 225    if existing:
 226        # If caller provided file_path, use it.
 227        if file_path and (existing.get("file_path") != file_path):
 228            sb = get_sb()
 229            sb.table("projects").update({
 230                "file_path": file_path,
 231                "last_accessed": _now_iso(),
 232            }).eq("id", existing["id"]).execute()
 233            existing["file_path"] = file_path
 234        # If file_path still missing, try computing from user base path.
 235        if not existing.get("file_path"):
 236            computed = compute_default_project_file_path(get_project_base_path_for_user(user_id), name)
 237            if computed:
 238                sb = get_sb()
 239                sb.table("projects").update({
 240                    "file_path": computed,
 241                    "last_accessed": _now_iso(),
 242                }).eq("id", existing["id"]).execute()
 243                existing["file_path"] = computed

 247            link_device_project_local_path_if_missing_for_user_devices(
 248                user_id=user_id,
 249                project_id=existing["id"],
 250                local_path=existing["file_path"],
 251            )
 252        return existing

 254    project_id = create_project(user_id, name, file_path=file_path)
 255    p = get_project_by_id(project_id)
 256    return p or {"id": project_id, "user_id": user_id, "name": name, "file_path": file_path}

 344def _safe_project_folder_name(name: str) -> str:
 345    # Convert a project name into a filesystem-friendly folder name.
 346    n = (name or "").strip()
 347    if not n:
 348        n = "Project"
 349    # Avoid path traversal and invalid separators.
 350    n = n.replace("\\", "-").replace("/", "-")
 351    n = re.sub(r"[^A-Za-z0-9._-]+", "-", n)
 352    n = n.strip("-") or "Project"
 353    return n

 356def compute_default_project_file_path(base_path: str | None, project_name: str) -> str | None:
 357    """Build a default local path by joining base_path with a sanitized project folder name."""
 358    if not base_path:
 359        return None
 360    base_path = str(base_path).strip()
 361    if not base_path or not os.path.isabs(base_path):
 362        return None
 363    safe_name = _safe_project_folder_name(project_name)
 364    return os.path.join(base_path, safe_name)

 987def claim_next_queued_command_for_user(*, user_id: str) -> dict | None:
 988    """Claim the oldest queued command across ALL of the user's projects."""
 989    sb = get_sb()
 990
 991    # Expire any commands stuck as "running" for too long.
 992    _expire_stale_running_commands(user_id)
 993
 994    # Find the oldest queued command belonging to this user.
 995    cmd_res = (
 996        sb.table("terminal_commands")
 997        .select("*")
 998        .eq("user_id", user_id)
 999        .eq("status", "queued")
1000        .order("created_at")
1001        .limit(1)
1002    )
1003    cmd = _execute_single(cmd_res)
1004    if not cmd:
1005        return None
1006
1007    # Claim it (update only if still queued).
1008    sb.table("terminal_commands").update({
1009        "status": "running",
1010        "started_at": _now_iso(),
1011    }).eq("id", cmd["id"]).eq("status", "queued").execute()
1012
1013    # Re-fetch to confirm claim.
1014    verified = _execute_single(sb.table("terminal_commands").select("*").eq("id", cmd["id"]))
1015    if not verified or verified.get("status") != "running":
1016        return None
1017

1019    session_id = verified.get("session_id")
1020    if session_id:
1021        sess_row = _execute_single(sb.table("terminal_sessions").select("project_id").eq("id", session_id))
1022        if sess_row:
1023            pid = sess_row.get("project_id")
1024            proj_row = _execute_single(sb.table("projects").select("file_path").eq("id", pid))
1025            if proj_row:
1026                verified["project_path"] = proj_row.get("file_path")
1027
1028    return _sidecar.enrich_command(verified)
